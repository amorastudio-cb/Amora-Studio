function criarControlePagamento(pedido) { const area = criarElemento("div", "payment-management"); area.appendChild( criarElemento("h4", "", "Controle financeiro") ); area.appendChild( criarElemento( "p", "payment-explanation", "Defina o orçamento e gere uma cobrança Pix automaticamente pelo Mercado Pago." ) ); const pagamentoStatus = pedido.pagamento_status || "Não solicitado"; let classeStatus = "not-requested"; if (pagamentoStatus === "Aguardando pagamento") { classeStatus = "waiting"; } else if (pagamentoStatus === "Confirmado") { classeStatus = "confirmed"; } area.appendChild( criarElemento( "span", "payment-status " + classeStatus, "Pagamento: " + pagamentoStatus ) ); const campo = criarElemento("div", "payment-field"); const label = criarElemento( "label", "", "Valor do orçamento (R$)" ); const input = document.createElement("input"); input.type = "number"; input.min = "0.01"; input.step = "0.01"; input.placeholder = "Ex.: 35,00"; input.value = pedido.valor_orcamento ?? ""; input.setAttribute("aria-label", "Valor do orçamento"); if (pagamentoStatus === "Confirmado") { input.disabled = true; } campo.appendChild(label); campo.appendChild(input); area.appendChild(campo); const botaoSalvar = criarElemento( "button", "payment-button secondary", "Salvar orçamento" ); botaoSalvar.type = "button"; if (pagamentoStatus === "Confirmado") { botaoSalvar.disabled = true; botaoSalvar.textContent = "Orçamento confirmado"; } botaoSalvar.addEventListener("click", async () => { const valorDigitado = input.value.trim(); const valor = Number(valorDigitado); if (!valorDigitado || !Number.isFinite(valor) || valor <= 0) { alert("Informe um valor válido para o orçamento."); input.focus(); return; } if (pagamentoStatus === "Confirmado") { alert("Este pagamento já foi confirmado. O orçamento não pode ser alterado."); return; } if (pedido.mercado_pago_payment_id) { alert( "Este pedido já possui uma cobrança Pix. Não altere o orçamento enquanto ela estiver ativa." ); return; } botaoSalvar.disabled = true; botaoSalvar.textContent = "Salvando..."; try { const { error } = await supabase .from("pedidos") .update({ valor_orcamento: valor }) .eq("id", pedido.id); if (error) throw error; alert("Orçamento salvo com sucesso!"); await carregarPedidos(); } catch (error) { console.error("Erro ao salvar orçamento:", error); alert( "Não foi possível salvar o orçamento. Verifique as permissões do banco." ); botaoSalvar.disabled = false; botaoSalvar.textContent = "Salvar orçamento"; } }); area.appendChild(botaoSalvar); const datas = criarElemento("div", "payment-dates"); datas.appendChild( criarElemento( "div", "", "Pix enviado: " + formatarDataHora(pedido.pix_enviado_em) ) ); datas.appendChild( criarElemento( "div", "", "Pagamento confirmado: " + formatarDataHora(pedido.pagamento_confirmado_em) ) ); if (pedido.mercado_pago_payment_id) { datas.appendChild( criarElemento( "div", "", "Código da cobrança: " + pedido.mercado_pago_payment_id ) ); } area.appendChild(datas); const acoes = criarElemento("div", "payment-actions"); if ( pagamentoStatus !== "Confirmado" && pagamentoStatus !== "Aguardando pagamento" ) { const botaoPix = criarElemento( "button", "payment-button", "Gerar cobrança Pix" ); botaoPix.type = "button"; botaoPix.addEventListener("click", async () => { const valorDigitado = input.value.trim(); const valor = Number(valorDigitado); if (!valorDigitado || !Number.isFinite(valor) || valor <= 0) { alert("Informe um valor válido para o orçamento."); input.focus(); return; } if (Number(pedido.valor_orcamento) !== valor) { alert("Primeiro, salve o orçamento atualizado."); return; } const confirmar = window.confirm( "Deseja gerar uma cobrança Pix de " + formatarMoeda(valor) + " para este pedido? O Mercado Pago criará o QR Code e o código Pix." ); if (!confirmar) return; botaoPix.disabled = true; botaoPix.textContent = "Gerando cobrança..."; try { const { data, error } = await supabase.functions.invoke( "gerar-pix", { body: { pedido_id: pedido.id } } ); if (error) { console.error("Erro ao chamar gerar-pix:", error); let mensagem = "Não foi possível gerar a cobrança Pix."; if (error.context) { try { const resposta = await error.context.json(); if (resposta?.erro) { mensagem = resposta.erro; } } catch (_) { // Mantém a mensagem padrão. } } throw new Error(mensagem); } if (!data?.sucesso || !data?.qr_code) { throw new Error( data?.erro || "O Mercado Pago não retornou os dados do Pix." ); } const { error: erroRegistro } = await supabase .from("pedidos") .update({ pix_enviado_em: pedido.pix_enviado_em || new Date().toISOString() }) .eq("id", pedido.id); if (erroRegistro) { console.error("Erro ao registrar data do Pix:", erroRegistro); alert( "A cobrança foi criada, mas não foi possível registrar a data no pedido. Atualize o painel antes de tentar novamente." ); await carregarPedidos(); return; } alert( "Cobrança Pix gerada com sucesso!\n\n" + "Valor: " + formatarMoeda(valor) + "\n\nO QR Code e o código Pix foram salvos. Na próxima etapa, vamos exibi-los na área do cliente." ); await carregarPedidos(); } catch (error) { console.error("Erro ao gerar cobrança Pix:", error); alert( error.message || "Ocorreu um erro ao gerar a cobrança Pix. Consulte o console para mais detalhes." ); botaoPix.disabled = false; botaoPix.textContent = "Gerar cobrança Pix"; } }); acoes.appendChild(botaoPix); } if (pagamentoStatus === "Aguardando pagamento") { acoes.appendChild( criarElemento( "p", "payment-explanation", "Cobrança Pix gerada. A confirmação será atualizada automaticamente quando o Mercado Pago confirmar o pagamento. Não é necessário confirmar manualmente." ) ); } if (pagamentoStatus === "Confirmado") { acoes.appendChild( criarElemento( "p", "payment-explanation", "Pagamento confirmado pelo sistema. Os arquivos finais poderão ser acessados pelo cliente." ) ); } area.appendChild(acoes); return area; }
